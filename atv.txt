export function multiplicar(numero1: number, numero2: number): number {
  return numero1 * numero2;
}

export function saudar(nome: string): string {
  return "Olá " + nome;
}

const numero1: number = Number(prompt("Número 1:"));
const numero2: number = Number(prompt("Número 2:"));
const nome: string = prompt("Digite seu nome:") ?? "";

console.log(`Resultado: ${multiplicar(numero1, numero2)}`);
console.log(saudar(nome));
